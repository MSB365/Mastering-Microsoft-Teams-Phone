Connect-MicrosoftTeams

New-CsTenantNetworkRegion -NetworkRegionID "Europe"
New-CsTenantNetworkSite   -NetworkSiteID "Zurich" -NetworkRegionID "Europe"

New-CsTenantNetworkSubnet -SubnetID "10.10.0.0" -MaskBits 24 -NetworkSiteID "Zurich"
New-CsTenantNetworkSubnet -SubnetID "10.10.1.0" -MaskBits 24 -NetworkSiteID "Zurich"

# Public egress address of the corporate network
New-CsTenantTrustedIPAddress -IPAddress "198.51.100.10" -MaskBits 32 `
    -Description "Zurich internet breakout"

Get-CsTenantNetworkSubnet | Select-Object SubnetID, MaskBits, NetworkSiteID
