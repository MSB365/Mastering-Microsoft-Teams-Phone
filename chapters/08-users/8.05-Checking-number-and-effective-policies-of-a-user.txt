$upn = "anna.meier@contoso.com"

Get-CsOnlineUser -Identity $upn |
    Select-Object EnterpriseVoiceEnabled, LineUri, FeatureTypes, UsageLocation

Get-CsPhoneNumberAssignment -AssignedPstnTargetId $upn |
    Select-Object TelephoneNumber, NumberType, AssignmentCategory, PstnAssignmentStatus

"OnlineVoiceRoutingPolicy","TenantDialPlan","TeamsCallingPolicy","CallingLineIdentity" |
    ForEach-Object {
        $a = Get-CsUserPolicyAssignment -Identity $upn -PolicyType $_
        [pscustomobject]@{ PolicyType = $_; Policy = $a.PolicyName }
    }
