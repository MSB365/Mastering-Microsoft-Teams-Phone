# Direct Routing number
Set-CsPhoneNumberAssignment -Identity "anna.meier@contoso.com" `
    -PhoneNumber "+41441234567" -PhoneNumberType DirectRouting

# Direct Routing number with extension
Set-CsPhoneNumberAssignment -Identity "ben.keller@contoso.com" `
    -PhoneNumber "+41441234500;ext=4501" -PhoneNumberType DirectRouting

# Operator Connect number
Set-CsPhoneNumberAssignment -Identity "carla.rossi@contoso.com" `
    -PhoneNumber "+41441234568" -PhoneNumberType OperatorConnect

# Calling Plan number with emergency location
Set-CsPhoneNumberAssignment -Identity "david.huber@contoso.com" `
    -PhoneNumber "+41441234569" -PhoneNumberType CallingPlan -LocationId $locId

# Voice-enabled without a number (e.g. Shared Calling)
Set-CsPhoneNumberAssignment -Identity "eva.frei@contoso.com" -EnterpriseVoiceEnabled $true
