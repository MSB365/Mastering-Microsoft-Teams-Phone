$users = Import-Csv .\wave-03.csv | Select-Object -ExpandProperty UPN

$op = New-CsBatchPolicyAssignmentOperation -PolicyType TeamsCallingPolicy `
    -PolicyName "CP-Standard" -Identity $users -OperationName "Wave 03 calling policy"

Get-CsBatchPolicyAssignmentOperation -OperationId $op
