$plan   = Import-Csv .\number-plan.csv          # columns: Number, UPN
$actual = Get-CsPhoneNumberAssignment | Select-Object TelephoneNumber, AssignedPstnTargetId

$diff = foreach ($row in $plan) {
    $a = $actual | Where-Object TelephoneNumber -eq $row.Number
    $assignedUpn = if ($a.AssignedPstnTargetId) {
        (Get-CsOnlineUser -Identity $a.AssignedPstnTargetId).UserPrincipalName } else { "" }
    if ($assignedUpn -ne $row.UPN) {
        [pscustomobject]@{ Number = $row.Number; Plan = $row.UPN; Tenant = $assignedUpn }
    }
}
$diff | Format-Table -AutoSize
