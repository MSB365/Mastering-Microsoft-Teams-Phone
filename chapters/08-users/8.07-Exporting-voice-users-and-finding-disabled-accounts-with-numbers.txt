$users = Get-CsOnlineUser -Filter 'EnterpriseVoiceEnabled -eq $true' -ResultSize Unlimited

$report = foreach ($u in $users) {
    [pscustomobject]@{
        UPN           = $u.UserPrincipalName
        Enabled       = $u.AccountEnabled
        LineUri       = $u.LineUri
        PhoneSystem   = $u.FeatureTypes -contains "PhoneSystem"
        VoiceRouting  = $u.OnlineVoiceRoutingPolicy
        DialPlan      = $u.TenantDialPlan
        CallingPolicy = $u.TeamsCallingPolicy
    }
}
$report | Export-Csv .\voice-users.csv -NoTypeInformation

# Disabled accounts that still hold a number
$report | Where-Object { -not $_.Enabled -and $_.LineUri }
