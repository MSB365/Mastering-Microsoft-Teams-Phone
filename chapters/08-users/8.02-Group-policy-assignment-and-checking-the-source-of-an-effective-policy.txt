Grant-CsOnlineVoiceRoutingPolicy -Group "voice-ch-intl@contoso.com" `
    -PolicyName "VRP-CH-International" -Rank 1
Grant-CsTenantDialPlan -Group "voice-ch-zurich@contoso.com" `
    -PolicyName "DP-CH-Zurich" -Rank 1

# Which policy is effective for a user, and where does it come from?
Get-CsUserPolicyAssignment -Identity "anna.meier@contoso.com" -PolicyType OnlineVoiceRoutingPolicy |
    Select-Object -ExpandProperty PolicySource
