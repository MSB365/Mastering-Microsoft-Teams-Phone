$ra = Get-CsOnlineApplicationInstance -Identity "aa-main@contoso.com"

New-CsTeamsSharedCallingRoutingPolicy -Identity "SC-Zurich" `
    -ResourceAccount $ra.ObjectId `
    -EmergencyNumbers @{ Add = "+41441234510", "+41441234511" }

New-CsOnlineVoiceRoutingPolicy -Identity "VRP-SharedCalling" -OnlinePstnUsages @()

$upn = "eva.frei@contoso.com"
Set-CsPhoneNumberAssignment -Identity $upn -EnterpriseVoiceEnabled $true
Grant-CsOnlineVoiceRoutingPolicy -Identity $upn -PolicyName "VRP-SharedCalling"
Grant-CsTeamsSharedCallingRoutingPolicy -Identity $upn -PolicyName "SC-Zurich"
