# Copilot during calls only: Copilot on, call transcription off
Set-CsTeamsCallingPolicy -Identity "CP-Copilot-LiveOnly" -Copilot Enabled `
    -AllowTranscriptionForCalling $false

# Copilot with a saved transcript required
Set-CsTeamsCallingPolicy -Identity "CP-Copilot-Transcript" -Copilot EnabledWithTranscript `
    -AllowTranscriptionForCalling $true

# No Copilot in calls
Set-CsTeamsCallingPolicy -Identity "CP-NoCopilot" -Copilot Disabled
