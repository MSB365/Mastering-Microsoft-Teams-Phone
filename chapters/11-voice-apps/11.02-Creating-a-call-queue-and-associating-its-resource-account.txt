$group = Get-MgGroup -Filter "displayName eq 'Sales Agents'"

$cq = New-CsCallQueue -Name "Sales" -LanguageId "de-CH" `
    -RoutingMethod LongestIdle -AgentAlertTime 30 -AllowOptOut $true `
    -ConferenceMode $true -DistributionLists @($group.Id)

New-CsOnlineApplicationInstanceAssociation -Identities @($ra.ObjectId) `
    -ConfigurationId $cq.Identity -ConfigurationType CallQueue
