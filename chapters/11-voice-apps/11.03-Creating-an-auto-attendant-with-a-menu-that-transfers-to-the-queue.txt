$cqTarget = New-CsAutoAttendantCallableEntity -Identity $ra.ObjectId -Type ApplicationEndpoint

$greeting = New-CsAutoAttendantPrompt -TextToSpeechPrompt "Willkommen bei Contoso."
$menuText = New-CsAutoAttendantPrompt -TextToSpeechPrompt "Für den Verkauf drücken Sie die 1."
$option1  = New-CsAutoAttendantMenuOption -Action TransferCallToTarget `
              -DtmfResponse Tone1 -CallTarget $cqTarget
$menu     = New-CsAutoAttendantMenu -Name "Hauptmenü" -MenuOptions @($option1) -Prompts @($menuText)
$flow     = New-CsAutoAttendantCallFlow -Name "Business hours" -Greetings @($greeting) -Menu $menu

$aa = New-CsAutoAttendant -Name "Contoso Main" -LanguageId "de-CH" `
    -TimeZoneId "W. Europe Standard Time" -DefaultCallFlow $flow
