# Requires an app registration with the application permission
# CallRecords.Read.All (admin consent required)
Connect-MgGraph -TenantId $TenantId -ClientId $AppId `
    -CertificateThumbprint $Thumbprint

$id  = "<call record id>"
$uri = "https://graph.microsoft.com/v1.0/communications/callRecords/$id" +
       "?`$expand=sessions(`$expand=segments)"

$record = Invoke-MgGraphRequest -Method GET -Uri $uri
$record.type                  # peerToPeer or groupCall
$record.modalities            # audio, video, screenSharing ...
$record.sessions.Count
