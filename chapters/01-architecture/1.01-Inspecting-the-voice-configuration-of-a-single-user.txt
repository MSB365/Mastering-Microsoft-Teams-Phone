Connect-MicrosoftTeams

$upn = "anna.meier@contoso.com"

Get-CsOnlineUser -Identity $upn |
    Select-Object DisplayName, UsageLocation, AccountEnabled,
        EnterpriseVoiceEnabled, LineUri, FeatureTypes,
        OnlineVoiceRoutingPolicy, TenantDialPlan,
        TeamsCallingPolicy, CallingLineIdentity

# Which dial plan is really in effect, and how does it normalize a number?
Get-CsEffectiveTenantDialPlan -Identity $upn |
    Test-CsEffectiveTenantDialPlan -DialedNumber "1234"
