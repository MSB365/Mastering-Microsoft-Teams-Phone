Connect-MgGraph -Scopes "Organization.Read.All"

Get-MgSubscribedSku |
    Select-Object SkuPartNumber, ConsumedUnits,
        @{ Name = "Enabled"; Expression = { $_.PrepaidUnits.Enabled } } |
    Sort-Object SkuPartNumber |
    Format-Table -AutoSize
