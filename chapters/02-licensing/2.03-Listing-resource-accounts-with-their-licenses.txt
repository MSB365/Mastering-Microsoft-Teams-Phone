Connect-MicrosoftTeams
Connect-MgGraph -Scopes "User.Read.All"

# Application IDs: auto attendant ce933385-9390-45d1-9512-c8d228074e07,
#                  call queue     11cd3e2e-fccb-42ad-ad00-878b93575e07
Get-CsOnlineApplicationInstance | ForEach-Object {
    $skus = (Get-MgUserLicenseDetail -UserId $_.ObjectId).SkuPartNumber
    [pscustomobject]@{
        Account     = $_.UserPrincipalName
        Type        = if ($_.ApplicationId -eq "ce933385-9390-45d1-9512-c8d228074e07") { "AA" } else { "CQ" }
        PhoneNumber = $_.PhoneNumber
        Licenses    = ($skus -join ", ")
    }
} | Format-Table -AutoSize
