Connect-MicrosoftTeams

$users = Get-CsOnlineUser -ResultSize Unlimited |
    Select-Object UserPrincipalName, LineUri, EnterpriseVoiceEnabled, FeatureTypes

# Phone number assigned, but no Teams Phone capability
$broken = $users | Where-Object {
    $_.LineUri -and ($_.FeatureTypes -notcontains "PhoneSystem") }

# Teams Phone capability, but not voice-enabled
$unused = $users | Where-Object {
    ($_.FeatureTypes -contains "PhoneSystem") -and -not $_.EnterpriseVoiceEnabled }

$broken | Export-Csv .\voice-broken.csv -NoTypeInformation
$unused | Export-Csv .\voice-unused.csv -NoTypeInformation
