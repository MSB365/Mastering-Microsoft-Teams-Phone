$uri   = "https://graph.microsoft.com/v1.0/admin/teams/userConfigurations"
$users = Get-GraphAll -Uri $uri

$users | Where-Object { $_.isEnterpriseVoiceEnabled } | ForEach-Object {
    [pscustomobject]@{
        UPN     = $_.userPrincipalName
        Type    = $_.accountType
        Numbers = ($_.telephoneNumbers.telephoneNumber -join ", ")
    }
} | Sort-Object UPN | Format-Table -AutoSize
