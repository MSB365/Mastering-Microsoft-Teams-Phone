$from = (Get-Date).ToUniversalTime().AddDays(-7).ToString("yyyy-MM-dd")
$to   = (Get-Date).ToUniversalTime().AddDays(1).ToString("yyyy-MM-dd")
$uri  = "https://graph.microsoft.com/v1.0/communications/callRecords/" +
        "getDirectRoutingCalls(fromDateTime=$from,toDateTime=$to)"

$calls  = Get-GraphAll -Uri $uri
$failed = $calls | Where-Object { -not $_.successfulCall }

"{0} calls, {1} failed ({2:P1})" -f $calls.Count, $failed.Count,
    ($(if ($calls.Count) { $failed.Count / $calls.Count } else { 0 }))

$failed | Group-Object trunkFullyQualifiedDomainName, finalSipCode, finalSipCodePhrase |
    Sort-Object Count -Descending |
    Select-Object Count, Name -First 15
