$uri = "https://graph.microsoft.com/v1.0/communications/callRecords/" +
       "getPstnCalls(fromDateTime=$from,toDateTime=$to)"

Get-GraphAll -Uri $uri | Group-Object callType | ForEach-Object {
    [pscustomobject]@{
        CallType = $_.Name
        Calls    = $_.Count
        Minutes  = [math]::Round(($_.Group | Measure-Object duration -Sum).Sum / 60)
    }
} | Sort-Object Minutes -Descending
