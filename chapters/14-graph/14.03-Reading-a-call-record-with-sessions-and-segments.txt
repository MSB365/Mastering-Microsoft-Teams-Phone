$id  = "<call record id>"
$uri = "https://graph.microsoft.com/v1.0/communications/callRecords/$id" +
       "?`$expand=sessions(`$expand=segments)"
$cr  = Invoke-MgGraphRequest -Method GET -Uri $uri -OutputType PSObject

"{0} call, {1} to {2}, modalities: {3}" -f $cr.type, $cr.startDateTime, $cr.endDateTime,
    ($cr.modalities -join ", ")

foreach ($s in $cr.sessions) {
    foreach ($seg in $s.segments) {
        "{0}  {1} -> {2}" -f $seg.startDateTime,
            $seg.caller.userAgent.headerValue, $seg.callee.userAgent.headerValue
    }
}
