$body = @{
    changeType         = "created"
    notificationUrl    = "https://voice-collector.contoso.com/api/notify"
    resource           = "/communications/callRecords"
    expirationDateTime = (Get-Date).ToUniversalTime().AddMinutes(4200).ToString("o")
    clientState        = "<random secret your endpoint checks>"
} | ConvertTo-Json

$sub = Invoke-MgGraphRequest -Method POST -Uri "https://graph.microsoft.com/v1.0/subscriptions" `
    -Body $body -ContentType "application/json" -OutputType PSObject
$sub.id, $sub.expirationDateTime
