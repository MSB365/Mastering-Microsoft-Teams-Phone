Connect-MgGraph -Identity -NoWelcome

$day  = (Get-Date).ToUniversalTime().Date.AddDays(-1)
$from = $day.ToString("yyyy-MM-dd")
$to   = $day.AddDays(1).ToString("yyyy-MM-dd")
$base = "https://graph.microsoft.com/v1.0/communications/callRecords"

$dr   = Get-GraphAll -Uri "$base/getDirectRoutingCalls(fromDateTime=$from,toDateTime=$to)"
$pstn = Get-GraphAll -Uri "$base/getPstnCalls(fromDateTime=$from,toDateTime=$to)"

$dr   | Export-Csv ".\dr-$from.csv"   -NoTypeInformation
$pstn | Export-Csv ".\pstn-$from.csv" -NoTypeInformation
