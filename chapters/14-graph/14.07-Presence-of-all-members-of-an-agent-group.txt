$group   = Get-MgGroup -Filter "displayName eq 'Sales Agents'"
$members = Get-MgGroupMember -GroupId $group.Id -All | Select-Object -ExpandProperty Id

$result = for ($i = 0; $i -lt $members.Count; $i += 650) {
    $ids  = $members[$i..([math]::Min($i + 649, $members.Count - 1))]
    $body = @{ ids = $ids } | ConvertTo-Json
    (Invoke-MgGraphRequest -Method POST `
        -Uri "https://graph.microsoft.com/v1.0/communications/getPresencesByUserId" `
        -Body $body -ContentType "application/json" -OutputType PSObject).value
}
$result | Group-Object availability | Select-Object Name, Count
