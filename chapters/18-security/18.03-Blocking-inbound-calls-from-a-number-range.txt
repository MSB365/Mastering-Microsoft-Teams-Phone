# Block a range of numbers that generates nuisance calls
New-CsInboundBlockedNumberPattern -Name "Block-Spam-Range" -Enabled $true `
    -Description "Nuisance calls, ticket INC0012345" -Pattern "^\+4190012\d{4}$"

# Make sure the blocking feature is active
Set-CsTenantBlockedCallingNumbers -Enabled $true

# Test before relying on it
Test-CsInboundBlockedNumberPattern -PhoneNumber "+41900121234"
