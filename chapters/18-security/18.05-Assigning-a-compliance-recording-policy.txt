New-CsTeamsComplianceRecordingPolicy -Identity "CR-Trading" -Enabled $true `
    -Description "Compliance recording for trading desk"

Set-CsTeamsComplianceRecordingPolicy -Identity "CR-Trading" -ComplianceRecordingApplications @(
    New-CsTeamsComplianceRecordingApplication -Parent "CR-Trading" `
        -Id "<object ID of the recorder resource account>" `
        -RequiredBeforeCallEstablishment $true -RequiredDuringCall $true
)

Grant-CsTeamsComplianceRecordingPolicy -Identity "trader01@contoso.com" -PolicyName "CR-Trading"
