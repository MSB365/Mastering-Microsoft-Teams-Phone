$from = (Get-Date).ToUniversalTime().AddDays(-1).ToString("yyyy-MM-dd")
$to   = (Get-Date).ToUniversalTime().ToString("yyyy-MM-dd")
$uri  = "https://graph.microsoft.com/v1.0/communications/callRecords/" +
        "getDirectRoutingCalls(fromDateTime=$from,toDateTime=$to)"
$tz   = [TimeZoneInfo]::FindSystemTimeZoneById("Europe/Zurich")

$suspicious = Get-GraphAll -Uri $uri | Where-Object {
    $local = [TimeZoneInfo]::ConvertTimeFromUtc(([datetime]$_.startDateTime).ToUniversalTime(), $tz)
    $_.callType -eq "ByotOut" -and $_.successfulCall -and
    $_.calleeNumber -notlike "+41*" -and
    ($local.Hour -lt 7 -or $local.Hour -ge 20 -or $local.DayOfWeek -in "Saturday","Sunday")
}

$suspicious | Group-Object userPrincipalName | Sort-Object Count -Descending |
    Select-Object Count, Name, @{ n = "Minutes"; e = { [math]::Round(($_.Group | Measure-Object duration -Sum).Sum / 60) } }
