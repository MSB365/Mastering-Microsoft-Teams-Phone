# Which restriction policies exist?
Get-CsOnlineDialOutPolicy | Select-Object Identity

# Domestic PSTN calls only, no dial-out from meetings, as the organization default
Grant-CsDialoutPolicy -Global -PolicyName "DialoutCPCDisabledPSTNDomestic"

# International calls for a group that needs them, for example sales
Grant-CsDialoutPolicy -Group "sg-voice-international@contoso.com" -Rank 10 `
    -PolicyName "DialoutCPCandPSTNInternational"
