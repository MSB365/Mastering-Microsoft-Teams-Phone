Set-CsPhoneNumberAssignment -Identity "ceo@contoso.com" `
    -PhoneNumber "+41441234999" -PhoneNumberType DirectRouting -AssignmentCategory Private

Get-CsPhoneNumberAssignment -AssignedPstnTargetId "ceo@contoso.com" |
    Select-Object TelephoneNumber, AssignmentCategory
