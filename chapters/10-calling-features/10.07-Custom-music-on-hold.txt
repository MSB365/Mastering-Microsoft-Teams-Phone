$bytes = [System.IO.File]::ReadAllBytes("C:\Audio\contoso-hold.mp3")
$file  = Import-CsOnlineAudioFile -ApplicationId "TenantGlobal" `
            -FileName "contoso-hold.mp3" -Content $bytes

New-CsTeamsCallHoldPolicy -Identity "Hold-Contoso" -AudioFileId $file.Id
Grant-CsTeamsCallHoldPolicy -Group "voice-users@contoso.com" -PolicyName "Hold-Contoso" -Rank 10
