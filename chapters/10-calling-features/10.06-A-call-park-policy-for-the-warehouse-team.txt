New-CsTeamsCallParkPolicy -Identity "CPP-Warehouse" -AllowCallPark $true `
    -PickupRangeStart 100 -PickupRangeEnd 199 -ParkTimeoutSeconds 600

Grant-CsTeamsCallParkPolicy -Group "warehouse@contoso.com" -PolicyName "CPP-Warehouse" -Rank 10
