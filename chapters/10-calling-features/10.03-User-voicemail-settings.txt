# Disable voicemail for a common area or lobby phone
Set-CsOnlineVoicemailUserSettings -Identity "lobby-phone@contoso.com" -VoicemailEnabled $false

# Let callers leave a message or transfer, e.g. to the reception
Set-CsOnlineVoicemailUserSettings -Identity "anna.meier@contoso.com" `
    -CallAnswerRules VoicemailWithTransferOption -TransferTarget "+41441234500"
