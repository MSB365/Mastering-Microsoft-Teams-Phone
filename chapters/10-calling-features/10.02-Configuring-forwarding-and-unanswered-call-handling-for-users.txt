# Forward all calls immediately to an external number
Set-CsUserCallingSettings -Identity "anna.meier@contoso.com" `
    -IsForwardingEnabled $true -ForwardingType Immediate `
    -ForwardingTargetType SingleTarget -ForwardingTarget "+41791234567"

# Send unanswered calls to voicemail after 20 seconds
Set-CsUserCallingSettings -Identity "ben.keller@contoso.com" `
    -IsUnansweredEnabled $true -UnansweredTargetType Voicemail `
    -UnansweredDelay 00:00:20

Get-CsUserCallingSettings -Identity "ben.keller@contoso.com"
