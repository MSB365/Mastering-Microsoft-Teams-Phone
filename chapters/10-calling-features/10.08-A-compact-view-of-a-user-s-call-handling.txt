$upn = "anna.meier@contoso.com"
$s = Get-CsUserCallingSettings -Identity $upn

[pscustomobject]@{
    Forwarding        = if ($s.IsForwardingEnabled) { "$($s.ForwardingType) -> $($s.ForwardingTarget)" } else { "off" }
    Unanswered        = if ($s.IsUnansweredEnabled) { "$($s.UnansweredTargetType) after $($s.UnansweredDelay)" } else { "off" }
    CallGroupOrder    = $s.CallGroupOrder
    CallGroupTargets  = ($s.CallGroupTargets -join ", ")
    Delegates         = ($s.Delegates.Id -join ", ")
    CallingPolicy     = (Get-CsUserPolicyAssignment -Identity $upn -PolicyType TeamsCallingPolicy).PolicyName
} | Format-List
