# All Cs cmdlets in the module, grouped by verb
Get-Command -Module MicrosoftTeams -Noun Cs* | Group-Object Verb | Select-Object Name, Count

# Find cmdlets for a topic
Get-Command -Module MicrosoftTeams -Noun *VoiceRoute*
Get-Command -Module MicrosoftTeams -Noun *PhoneNumber*

# Full help with examples
