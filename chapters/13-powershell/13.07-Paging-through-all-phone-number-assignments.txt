$all = @(); $skip = 0; $page = 500
do {
    $batch = Get-CsPhoneNumberAssignment -Top $page -Skip $skip
    $all  += $batch
    $skip += $page
} while ($batch.Count -eq $page)
"{0} numbers" -f $all.Count
