$users = Get-CsOnlineUser -Properties Identity, UserPrincipalName -ResultSize Unlimited
$byId  = @{}
foreach ($u in $users) { $byId[[string]$u.Identity] = $u.UserPrincipalName }

Get-CsPhoneNumberAssignment | ForEach-Object {
    [pscustomobject]@{
        Number = $_.TelephoneNumber
        Type   = $_.NumberType
        User   = $byId[[string]$_.AssignedPstnTargetId]
    }
}
