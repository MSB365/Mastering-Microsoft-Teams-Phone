# Install for the current user (no administrator rights needed)
Install-Module -Name MicrosoftTeams -Scope CurrentUser

# Update later; close all sessions that have the module loaded first
Update-Module -Name MicrosoftTeams

# Which versions are installed?
Get-InstalledModule -Name MicrosoftTeams -AllVersions | Select-Object Name, Version
