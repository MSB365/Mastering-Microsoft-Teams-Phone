# Voice-enabled users only, with a few properties
Get-CsOnlineUser -Filter 'EnterpriseVoiceEnabled -eq $true' `
    -Properties UserPrincipalName, LineUri, FeatureTypes -ResultSize Unlimited

# Users with the Teams Phone capability
Get-CsOnlineUser -Filter "FeatureTypes -contains 'PhoneSystem'" -ResultSize Unlimited

# Who has a specific number?
Get-CsOnlineUser -Filter {LineURI -eq "tel:+41441234567"}

# Resource accounts only
Get-CsOnlineUser -AccountType ResourceAccount
