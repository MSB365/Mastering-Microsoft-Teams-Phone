# Direct Routing variant: enable Location-Based Routing for the service center sites
Set-CsTenantNetworkSite -Identity "IN-PUNE-SC1" -EnableLocationBasedRouting $true
Set-CsTenantNetworkSite -Identity "IN-PUNE-SC2" -EnableLocationBasedRouting $true

# Verify subnet coverage: every subnet of the site must be defined
Get-CsTenantNetworkSubnet | Where-Object NetworkSiteID -like "IN-PUNE-*" |
    Select-Object SubnetID, MaskBits, NetworkSiteID
