New-CsTeamsIPPhonePolicy -Identity "IPP-HotDesk" -SignInMode CommonAreaPhoneSignIn `
    -AllowHotDesking $true -HotDeskingIdleTimeoutInMinutes 120

Grant-CsTeamsIPPhonePolicy -Identity "desk-3-12@contoso.com" -PolicyName "IPP-HotDesk"
