$ra = Get-CsOnlineApplicationInstance -Identity "aa-main@contoso.com"

New-CsCallingLineIdentity -Identity "CLI-MainNumber" `
    -Description "Show main number" `
    -CallingIDSubstitute Resource -ResourceAccount $ra.ObjectId `
    -EnableUserOverride $false

Grant-CsCallingLineIdentity -Identity "sales.user@contoso.com" -PolicyName "CLI-MainNumber"
