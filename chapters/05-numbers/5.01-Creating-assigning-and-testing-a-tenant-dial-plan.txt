Connect-MicrosoftTeams

$rule = New-CsVoiceNormalizationRule -Parent Global -Name "ZH-Ext-4xxx" `
    -Description "Zurich four-digit extensions" `
    -Pattern '^(4\d{3})$' -Translation '+4144123$1' -InMemory

New-CsTenantDialPlan -Identity "DP-CH-Zurich" -NormalizationRules @{ Add = $rule }

Grant-CsTenantDialPlan -Identity "anna.meier@contoso.com" -PolicyName "DP-CH-Zurich"

# Test the effective dial plan of the user
Get-CsEffectiveTenantDialPlan -Identity "anna.meier@contoso.com" |
    Test-CsEffectiveTenantDialPlan -DialedNumber "4567"
