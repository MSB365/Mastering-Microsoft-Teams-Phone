Connect-MicrosoftTeams

$log = @()
Import-Csv .\wave-03.csv | ForEach-Object {
    $row = $_
    try {
        Set-CsPhoneNumberAssignment -Identity $row.UPN -PhoneNumber $row.Number `
            -PhoneNumberType $row.NumberType -ErrorAction Stop
        if ($row.VoiceRoutingPolicy) {
            Grant-CsOnlineVoiceRoutingPolicy -Identity $row.UPN -PolicyName $row.VoiceRoutingPolicy
        }
        if ($row.DialPlan) {
            Grant-CsTenantDialPlan -Identity $row.UPN -PolicyName $row.DialPlan
        }
        $log += [pscustomobject]@{ UPN = $row.UPN; Number = $row.Number; Result = "OK" }
    }
    catch {
        $log += [pscustomobject]@{ UPN = $row.UPN; Number = $row.Number; Result = $_.Exception.Message }
    }
}
$log | Export-Csv .\wave-03-result.csv -NoTypeInformation
