Grant-CsOnlineVoiceRoutingPolicy -Identity "anna.meier@contoso.com" `
    -PolicyName "VRP-CH-International"

# Assign to a group with a rank (lower rank = higher priority)
Grant-CsOnlineVoiceRoutingPolicy -Group "voice-ch-intl@contoso.com" `
    -PolicyName "VRP-CH-International" -Rank 10
