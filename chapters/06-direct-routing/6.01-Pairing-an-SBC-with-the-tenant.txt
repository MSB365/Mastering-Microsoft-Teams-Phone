Connect-MicrosoftTeams

New-CsOnlinePSTNGateway -Fqdn "sbc01.contoso.com" `
    -SipSignalingPort 5061 `
    -MaxConcurrentSessions 100 `
    -ForwardCallHistory $true `
    -Enabled $true

Get-CsOnlinePSTNGateway -Identity "sbc01.contoso.com"
