New-CsOnlineVoiceRoute -Identity "CH-National" `
    -NumberPattern '^\+41\d{9}$' `
    -OnlinePstnGatewayList "sbc01.contoso.com", "sbc02.contoso.com" `
    -Priority 1 -OnlinePstnUsages "CH-National"

New-CsOnlineVoiceRoute -Identity "International" `
    -NumberPattern '^\+\d+$' `
    -OnlinePstnGatewayList "sbc01.contoso.com", "sbc02.contoso.com" `
    -Priority 2 -OnlinePstnUsages "International"
