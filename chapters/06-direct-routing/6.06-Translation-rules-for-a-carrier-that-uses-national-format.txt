New-CsTeamsTranslationRule -Identity "CH-E164-to-National" `
    -Pattern '^\+41(\d{9})$' -Translation '0$1'

New-CsTeamsTranslationRule -Identity "CH-0041-to-E164" `
    -Pattern '^0041(\d{9})$' -Translation '+41$1'

Set-CsOnlinePSTNGateway -Identity "sbc01.contoso.com" `
    -OutboundPstnNumberTranslationRules @{ Add = "CH-E164-to-National" } `
    -InboundPstnNumberTranslationRules  @{ Add = "CH-0041-to-E164" }
