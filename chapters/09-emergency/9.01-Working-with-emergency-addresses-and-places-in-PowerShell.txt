# List addresses and their places
Get-CsOnlineLisCivicAddress | Select-Object CivicAddressId, CompanyName, City, Description
Get-CsOnlineLisLocation    | Select-Object LocationId, CivicAddressId, Location, City

# Add a place (floor) to an existing address
$addr = Get-CsOnlineLisCivicAddress | Where-Object { $_.Description -eq "Zurich HQ" }
New-CsOnlineLisLocation -CivicAddressId $addr.CivicAddressId -Location "3rd floor"
