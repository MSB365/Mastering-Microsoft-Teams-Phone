New-CsTeamsEmergencyCallingPolicy -Identity "ECP-Zurich" `
    -NotificationGroup "security-zurich@contoso.com" `
    -NotificationMode NotificationOnly

# Assign to a network site (dynamic) and/or to users (static)
Set-CsTenantNetworkSite -Identity "Zurich" -EmergencyCallingPolicy "ECP-Zurich"
Grant-CsTeamsEmergencyCallingPolicy -Identity "anna.meier@contoso.com" -PolicyName "ECP-Zurich"
