# Export selected Teams voice resources as a DSC configuration
Export-M365DSCConfiguration -Components @("TeamsVoiceRoute", "TeamsVoiceRoutingPolicy", "TeamsCallingPolicy") `
    -ApplicationId $AppId -TenantId "contoso.onmicrosoft.com" `
    -CertificateThumbprint $Thumbprint -Path .\m365dsc

# Compare two exports and produce a report
New-M365DSCDeltaReport -Source .\m365dsc-old\M365TenantConfig.ps1 `
    -Destination .\m365dsc\M365TenantConfig.ps1 -OutputPath .\delta.html
