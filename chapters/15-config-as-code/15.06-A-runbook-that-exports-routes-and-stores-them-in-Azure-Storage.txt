Connect-AzAccount -Identity | Out-Null
Connect-MicrosoftTeams -Identity | Out-Null

$date = Get-Date -Format "yyyy-MM-dd"
$tmp  = Join-Path $env:TEMP "export-$date"
New-Item -ItemType Directory -Path $tmp -Force | Out-Null

Get-CsOnlineVoiceRoute | ConvertTo-Json -Depth 10 | Set-Content "$tmp\VoiceRoutes.json"
# ... further objects as in Listing 13.10 ...

$ctx = New-AzStorageContext -StorageAccountName "stvoiceops" -UseConnectedAccount
Get-ChildItem $tmp -File | ForEach-Object {
    Set-AzStorageBlobContent -File $_.FullName -Container "exports" `
        -Blob "$date/$($_.Name)" -Context $ctx -Force | Out-Null
}
