$payload = @{
    title   = "Voice configuration drift detected"
    details = ($drift | ForEach-Object { "$($_.Change): $($_.Object)" }) -join "`n"
} | ConvertTo-Json

Invoke-RestMethod -Method Post -Uri $WorkflowUrl -Body $payload -ContentType "application/json"
