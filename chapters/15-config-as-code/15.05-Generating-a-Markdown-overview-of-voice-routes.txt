$routes = Get-Content .\export\VoiceRoutes.json -Raw | ConvertFrom-Json

$lines  = @("| Route | Priority | Pattern | Usages | Gateways |", "|---|---|---|---|---|")
$lines += $routes | Sort-Object Priority | ForEach-Object {
    "| {0} | {1} | ``{2}`` | {3} | {4} |" -f $_.Identity, $_.Priority, $_.NumberPattern,
        ($_.OnlinePstnUsages -join ", "), ($_.OnlinePstnGatewayList -join ", ")
}
$lines | Set-Content .\docs\voice-routes.md -Encoding UTF8
