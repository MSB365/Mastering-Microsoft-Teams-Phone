$normal = 0, 480, 486, 487, 603   # answered or normal user outcomes
$from = (Get-Date).ToUniversalTime().AddHours(-1).ToString("yyyy-MM-ddTHH:mm:ssZ")
$to   = (Get-Date).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ssZ")
$uri  = "https://graph.microsoft.com/v1.0/communications/callRecords/" +
        "getDirectRoutingCalls(fromDateTime=$from,toDateTime=$to)"

Get-GraphAll -Uri $uri | Group-Object trunkFullyQualifiedDomainName | ForEach-Object {
    $ok    = @($_.Group | Where-Object { $_.successfulCall -or $normal -contains $_.finalSipCode }).Count
    $ratio = $ok / $_.Count
    [pscustomobject]@{ Trunk = $_.Name; Calls = $_.Count; Ratio = [math]::Round($ratio, 3) }
} | Where-Object { $_.Calls -ge 20 -and $_.Ratio -lt 0.95 }   # thresholds from your baseline
