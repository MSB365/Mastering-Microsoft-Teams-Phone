$sites = @{
    "ZRH-HQ"  = @{ City = "Zurich"; Zip = "8001"; Country = "CH"; Region = "EMEA" }
    "BER-OFF" = @{ City = "Berlin"; Zip = "10115"; Country = "DE"; Region = "EMEA" }
}

Get-CsTenantNetworkSubnet | ForEach-Object {
    $s = $sites[$_.NetworkSiteID]
    if (-not $s) { Write-Warning "No site data for $($_.NetworkSiteID)"; return }
    ($_.SubnetID, "$($_.NetworkSiteID)/$($_.SubnetID)", $_.MaskBits, $_.NetworkSiteID,
     "Contoso", "Office", "", $s.City, $s.Zip, $s.Country, "", $s.Region, 1, 0, 0) -join "`t"
} | Set-Content .\cqd-building.tsv -Encoding UTF8
