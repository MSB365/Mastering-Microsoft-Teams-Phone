# Requires the application permission CallRecords.Read.All
$from = (Get-Date).AddDays(-7).ToString("yyyy-MM-dd")
$to   = (Get-Date).ToString("yyyy-MM-dd")
$uri  = "https://graph.microsoft.com/v1.0/communications/callRecords/" +
        "getPstnCalls(fromDateTime=$from,toDateTime=$to)"

$calls = (Invoke-MgGraphRequest -Method GET -Uri $uri -OutputType PSObject).value
$calls | Group-Object callType | Select-Object Name, Count,
    @{ n = "Minutes"; e = { [math]::Round(($_.Group | Measure-Object duration -Sum).Sum / 60) } }
