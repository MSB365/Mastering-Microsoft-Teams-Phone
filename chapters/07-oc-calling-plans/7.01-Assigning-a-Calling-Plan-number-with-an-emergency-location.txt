Connect-MicrosoftTeams

# Find the emergency location (place) to use
$loc = Get-CsOnlineLisLocation | Where-Object { $_.City -eq "Zurich" } |
       Select-Object -First 1

Set-CsPhoneNumberAssignment -Identity "anna.meier@contoso.com" `
    -PhoneNumber "+41441234567" -PhoneNumberType CallingPlan `
    -LocationId $loc.LocationId
