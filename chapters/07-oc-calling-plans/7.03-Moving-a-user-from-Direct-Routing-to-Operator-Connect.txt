# Check the current assignment
Get-CsPhoneNumberAssignment -AssignedPstnTargetId "anna.meier@contoso.com"

# Remove the Direct Routing number and the voice routing policy
Remove-CsPhoneNumberAssignment -Identity "anna.meier@contoso.com" `
    -PhoneNumber "+41441234567" -PhoneNumberType DirectRouting
Grant-CsOnlineVoiceRoutingPolicy -Identity "anna.meier@contoso.com" -PolicyName $null

# After the operator has uploaded the number
Set-CsPhoneNumberAssignment -Identity "anna.meier@contoso.com" `
    -PhoneNumber "+41441234567" -PhoneNumberType OperatorConnect
