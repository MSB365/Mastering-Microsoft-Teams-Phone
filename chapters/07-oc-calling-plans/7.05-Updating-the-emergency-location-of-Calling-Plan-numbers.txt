$newLoc = Get-CsOnlineLisLocation | Where-Object { $_.Location -eq "Bern, 2nd floor" }

# Change the emergency location of a single number
Set-CsPhoneNumberAssignment -PhoneNumber "+41441234567" -LocationId $newLoc.LocationId

# Change it for all numbers of the users in a CSV file
Import-Csv .\moved-users.csv | ForEach-Object {
    $pn = Get-CsPhoneNumberAssignment -AssignedPstnTargetId $_.UPN -NumberType CallingPlan
    if ($pn) {
        Set-CsPhoneNumberAssignment -PhoneNumber $pn.TelephoneNumber -LocationId $newLoc.LocationId
    }
}
