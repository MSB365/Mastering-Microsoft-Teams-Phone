Get-CsPhoneNumberAssignment |
    Select-Object TelephoneNumber, NumberType, Capability,
        PstnAssignmentStatus, AssignedPstnTargetId |
    Sort-Object NumberType, TelephoneNumber |
    Format-Table -AutoSize
