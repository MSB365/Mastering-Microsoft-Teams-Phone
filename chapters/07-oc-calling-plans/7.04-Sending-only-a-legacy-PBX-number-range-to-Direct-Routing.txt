# Only calls to the legacy PBX range go to the SBC; everything else uses the Calling Plan
Set-CsOnlinePstnUsage -Identity Global -Usage @{ Add = "Legacy-PBX" }
New-CsOnlineVoiceRoute -Identity "Legacy-PBX-Range" -NumberPattern '^\+4144123[5-9]\d{3}$' `
    -OnlinePstnGatewayList "sbc01.contoso.com" -Priority 1 -OnlinePstnUsages "Legacy-PBX"
New-CsOnlineVoiceRoutingPolicy -Identity "VRP-CallingPlan-Plus-PBX" -OnlinePstnUsages "Legacy-PBX"
